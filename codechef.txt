#include <stdio.h>
#include <stdlib.h>

// Define the Trie Node structure
typedef struct TrieNode {
    struct TrieNode* children[2];
} TrieNode;

// Helper function to create a new Trie Node
TrieNode* createNode() {
    TrieNode* node = (TrieNode*)malloc(sizeof(TrieNode));
    node->children[0] = NULL;
    node->children[1] = NULL;
    return node;
}

// Insert a number into the Trie bit by bit (from MSB to LSB)
void insert(TrieNode* root, int num) {
    TrieNode* curr = root;
    for (int i = 30; i >= 0; i--) {
        int bit = (num >> i) & 1;
        if (!curr->children[bit]) {
            curr->children[bit] = createNode();
        }
        curr = curr->children[bit];
    }
}

// Find the maximum XOR possible for a given number against the Trie
int getMaxXOR(TrieNode* root, int num) {
    TrieNode* curr = root;
    int maxXOR = 0;
    for (int i = 30; i >= 0; i--) {
        int bit = (num >> i) & 1;
        int toggledBit = 1 - bit; // We want the opposite bit to maximize XOR
        
        // If the opposite bit exists, go there to get a 1 in XOR result
        if (curr->children[toggledBit]) {
            maxXOR |= (1 << i);
            curr = curr->children[toggledBit];
        } else {
            // Otherwise, we are forced to take the same bit (results in 0 for this position)
            curr = curr->children[bit];
        }
    }
    return maxXOR;
}

// Free the allocated memory for the Trie
void freeTrie(TrieNode* root) {
    if (!root) return;
    freeTrie(root->children[0]);
    freeTrie(root->children[1]);
    free(root);
}

// Main solution function
int findMaximumXOR(int* nums, int numsSize) {
    if (numsSize < 2) return 0;
    
    TrieNode* root = createNode();
    int globalMaxXOR = 0;
    
    // Insert the first number
    insert(root, nums[0]);
    
    // For each subsequent number, find its max XOR with existing numbers, then insert it
    for (int i = 1; i < numsSize; i++) {
        int currentMax = getMaxXOR(root, nums[i]);
        if (currentMax > globalMaxXOR) {
            globalMaxXOR = currentMax;
        }
        insert(root, nums[i]);
    }
    
    freeTrie(root);
    return globalMaxXOR;
}
