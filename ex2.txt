    printf("Final Codeword (with errors): %s\n", codeword);
    printf("=======================================================\n");

int searchURL(char url[])
{
    int i;
    for(i = 0; i < total; i++)
    {
        if(strcmp(url, table[i].url) == 0)
            return i;
    }
    return -1;
}

int readTextFromFile(char filename[], char text[])
{
    FILE *f = fopen(filename, "r");
    if(f == NULL) return 0;

    if(fgets(text, MAX_TEXT, f) == NULL) {
        fclose(f);
        return 0;
    }

    text[strcspn(text, "")] = '\0';
    fclose(f);
    return 1;
}

void decimalToBinary(int num, char binary[])
{
    int i;
    for(i = 7; i >= 0; i--)
        binary[7 - i] = ((num >> i) & 1) + '0';
    binary[8] = '\0';
}

void ipToBinary(char ip[], char binary[])
{
    int a, b, c, d;
    char temp[9];
    binary[0] = '\0';
    sscanf(ip, "%d.%d.%d.%d", &a, &b, &c, &d);
    decimalToBinary(a, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(b, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(c, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(d, temp); strcat(binary, temp);
}

void macToBinary(char mac[], char binary[])
{
    unsigned int a, b, c, d, e, f;
    char temp[9];
    binary[0] = '\0';
    sscanf(mac, "%x:%x:%x:%x:%x:%x", &a, &b, &c, &d, &e, &f);
    decimalToBinary(a, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(b, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(c, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(d, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(e, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(f, temp); strcat(binary, temp);
}

void textToBinary(char text[], char binary[])
{
    int i;
    char temp[9];
    binary[0] = '\0';
    for(i = 0; text[i] != '\0'; i++)
    {
        decimalToBinary((int)text[i], temp);
        strcat(binary, temp);
    }
}

int randomPort(void)
{
    return 1024 + rand() % 64512;
}

void runCRC(char dataBits[], char poly[], char remainderOut[], int higherOrderBits)
{
    int dataLen = strlen(dataBits);
    int polyLen = strlen(poly);
    int totalEvalLen, i, j;
    char *temp;

    temp = (char *)malloc((dataLen + polyLen + 5) * sizeof(char));
    strcpy(temp, dataBits);

    for(i = 0; i < higherOrderBits; i++) {
        temp[dataLen + i] = '0';
    }
    temp[dataLen + higherOrderBits] = '\0';

    totalEvalLen = strlen(temp);

    for (i = 0; i <= totalEvalLen - polyLen; i++) {
        if (temp[i] == '1') {
            for (j = 0; j < polyLen; j++) {
                temp[i + j] = (temp[i + j] == poly[j]) ? '0' : '1';
            }
        }
    }

    strncpy(remainderOut, &temp[totalEvalLen - higherOrderBits], higherOrderBits);
    remainderOut[higherOrderBits] = '\0';

    free(temp);
}

#include <stdio.h>
#include <string.h>

void charTo7Bit(char ch, char bin[])
{
    int i;
    for(i = 6; i >= 0; i--)
    {
        if(ch % 2 == 0)
            bin[i] = '0';
        else
            bin[i] = '1';

        ch = ch / 2;
    }
    bin[7] = '\0';
}

char getEvenParity(char bin[])
{
    int i;
    int ones = 0;

    for(i = 0; i < 7; i++)
    {
        if(bin[i] == '1')
            ones++;
    }

    if(ones % 2 == 0)
        return '0';
    else
        return '1';
}

int countOnes(char data[])
{
    int i;
    int count = 0;

    for(i = 0; i < 8; i++)
    {
        if(data[i] == '1')
            count++;
    }

    return count;
}

int main()
{
    char message[100];
    char blocks[100][9];
    int i, j;

    printf("Enter a message: ");
    fgets(message, sizeof(message), stdin);

    int len = strlen(message);
    if(len > 0 && message[len - 1] == '\n')
    {
        message[len - 1] = '\0';
        len--;
    }

    printf("\n--- Binary Table ---\n\n");

    for(i = 0; i < len; i++)

        char sevenBit[8];

        charTo7Bit(message[i], sevenBit);

        for(j = 0; j < 7; j++)
        {
            blocks[i][j] = sevenBit[j];
        }

        char parityBit = getEvenParity(sevenBit);
        blocks[i][7] = parityBit;
        blocks[i][8] = '\0';

        printf("  %d\t %c\t%s\t\t%c\t\t%s\n", i + 1, message[i], sevenBit, parityBit, blocks[i]);
    }

    char choice;
    printf("\nDo you want to change any bits? (Y/N): ");
    scanf(" %c", &choice);

    if(choice == 'Y' || choice == 'y')
    {
        int bitsToModify;
        printf("How many bits do you want to change in each block?: ");
        scanf("%d", &bitsToModify);

        for(i = 0; i < len; i++)
        {
            printf("\n--- Block %d ('%c') ---\n", i + 1, message[i]);

            for(j = 0; j < bitsToModify; j++)
            {
                int bitPos;
                char newValue;

                printf("Enter position (1 to 8): ");

                blocks[i][bitPos - 1] = newValue;
            }
        }

        printf("\n--- Updated Blocks ---\n");
        for(i = 0; i < len; i++)
        {
            printf("Block %d: %s\n", i + 1, blocks[i]);
        }

        printf("\n--- Checking for Errors ---\n");
        int hasError = 0;

        for(i = 0; i < len; i++)

            {
                printf("Block %d: No error found.\n", i + 1);
            }
        }

        if(hasError == 0)

            printf("\nAll blocks are fine.\n");
        }
    }
    else
    {
        printf("\nNo changes made.\n");
    }

    return 0;
}
