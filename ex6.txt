#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8085
#define MAX_SIZE 2000

int receive_all(int socket_fd, char *buffer, int length)
{
    int total = 0;

    while (total < length)
    {
        int received = recv(socket_fd,
                            buffer + total,
                            length - total,
                            0);

        if (received <= 0)
        {
            return -1;
        }

        total += received;
    }

    return total;
}

int toDecimal(char byte[])
{
    int value = 0;

    for (int i = 0; i < 8; i++)
    {
        value = value * 2 + (byte[i] - '0');
    }

    return value;
}

void display_binary(char data[])
{
    for (int i = 0; data[i] != '\0'; i++)
    {
        printf("%c", data[i]);

        if ((i + 1) % 8 == 0)
        {
            printf(" ");
        }
    }

    printf("\n");
}

/*
   Remove byte stuffing

   ESC  = 01111101
   FLAG = 01111110
*/
void byteUnstuffing(char stuffed[], char unstuffed[])
{
    int i;
    int k = 0;

    for (i = 0; stuffed[i] != '\0'; i += 8)
    {
        char byte[9];

        strncpy(byte, stuffed + i, 8);
        byte[8] = '\0';

        if (strcmp(byte, "01111101") == 0)
        {
            i += 8;

            if (stuffed[i] == '\0')
            {
                break;
            }

            strncpy(unstuffed + k, stuffed + i, 8);
            k += 8;
        }
        else
        {
            strcpy(unstuffed + k, byte);
            k += 8;
        }
    }

    unstuffed[k] = '\0';
}

void calculateChecksum(char binary[], char checksum[])
{
    int sum = 0;
    char byte[9];

    for (int i = 0; binary[i] != '\0'; i += 8)
    {
        strncpy(byte, binary + i, 8);
        byte[8] = '\0';

        sum += toDecimal(byte);

        /*
           End-around carry
        */
        while (sum > 255)
        {
            sum = (sum & 255) + (sum >> 8);
        }
    }

    /*
       One's complement
    */
    sum = 255 - sum;

    for (int i = 7; i >= 0; i--)
    {
        checksum[i] = (sum % 2) + '0';
        sum = sum / 2;
    }

    checksum[8] = '\0';
}

void binaryToText(char binary[], char text[])
{
    int k = 0;

    for (int i = 0; binary[i] != '\0'; i += 8)
    {
        char byte[9];

        strncpy(byte, binary + i, 8);
        byte[8] = '\0';

        text[k++] = (char)toDecimal(byte);
    }

    text[k] = '\0';
}

int main()
{
    int server_fd;
    int client_fd;
    int option = 1;

    int data_length;
    int network_length;

    struct sockaddr_in server;
    struct sockaddr_in client;

    socklen_t client_length;

    char received_data[MAX_SIZE];
    char unstuffed_data[MAX_SIZE];

    char original_data[MAX_SIZE];
    char received_checksum[9];
    char calculated_checksum[9];

    char received_text[MAX_SIZE];

    server_fd = socket(AF_INET, SOCK_STREAM, 0);

    if (server_fd < 0)
    {
        perror("Socket creation failed");
        return 1;
    }

    if (setsockopt(server_fd,
                   SOL_SOCKET,
                   SO_REUSEADDR,
                   &option,
                   sizeof(option)) < 0)
    {
        perror("setsockopt failed");
        close(server_fd);
        return 1;
    }

    memset(&server, 0, sizeof(server));

    server.sin_family = AF_INET;
    server.sin_addr.s_addr = htonl(INADDR_ANY);
    server.sin_port = htons(PORT);

    if (bind(server_fd,
             (struct sockaddr *)&server,
             sizeof(server)) < 0)
    {
        perror("Bind failed");
        close(server_fd);
        return 1;
    }

    if (listen(server_fd, 5) < 0)
    {
        perror("Listen failed");
        close(server_fd);
        return 1;
    }

    printf("====================================\n");
    printf("          CHECKSUM SERVER\n");
    printf("====================================\n");
    printf("Server waiting for client...\n");

    client_length = sizeof(client);

    client_fd = accept(server_fd,
                       (struct sockaddr *)&client,
                       &client_length);

    if (client_fd < 0)
    {
        perror("Accept failed");
        close(server_fd);
        return 1;
    }

    printf("Client connected successfully.\n");

    /*
       Receive data length
    */
    if (receive_all(client_fd,
                    (char *)&network_length,
                    sizeof(network_length)) < 0)
    {
        printf("Failed to receive data length.\n");
        close(client_fd);
        close(server_fd);
        return 1;
    }

    data_length = ntohl(network_length);

    if (data_length <= 0 || data_length >= MAX_SIZE)
    {
        printf("Invalid data length.\n");
        close(client_fd);
        close(server_fd);
        return 1;
    }

    /*
       Receive stuffed data
    */
    if (receive_all(client_fd,
                    received_data,
                    data_length) < 0)
    {
        printf("Failed to receive data.\n");
        close(client_fd);
        close(server_fd);
        return 1;
    }

    received_data[data_length] = '\0';

    printf("\nStuffed data received:\n");
    display_binary(received_data);

    /*
       Remove byte stuffing
    */
    byteUnstuffing(received_data, unstuffed_data);

    printf("\nAfter byte unstuffing:\n");
    display_binary(unstuffed_data);

    int unstuffed_length = strlen(unstuffed_data);

    /*
       Original data + 8-bit checksum
    */
    if (unstuffed_length < 16 ||
        unstuffed_length % 8 != 0)
    {
        printf("\nInvalid packet format.\n");

        close(client_fd);
        close(server_fd);
        return 1;
    }

    /*
       Last 8 bits are received checksum
    */
    int original_length = unstuffed_length - 8;

    strncpy(original_data,
            unstuffed_data,
            original_length);

    original_data[original_length] = '\0';

    strncpy(received_checksum,
            unstuffed_data + original_length,
            8);

    received_checksum[8] = '\0';

    /*
       Calculate checksum again
    */
    calculateChecksum(original_data, calculated_checksum);

    printf("\nReceived checksum  : %s\n",
           received_checksum);

    printf("Calculated checksum: %s\n",
           calculated_checksum);

    /*
       Verify checksum
    */
    if (strcmp(received_checksum,
               calculated_checksum) != 0)
    {
        printf("\nChecksum verification failed.\n");
        printf("Error detected in received data.\n");
        printf("Received text will not be displayed.\n");
    }
    else
    {
        binaryToText(original_data, received_text);

        printf("\nChecksum verification successful.\n");
        printf("No error detected.\n");
        printf("Received text: %s\n", received_text);
    }

    close(client_fd);
    close(server_fd);

    printf("\nServer closed\n");

    return 0;
}

int create_socket(void)
{
    int ls;

    ls = socket(AF_INET, SOCK_STREAM, 0);

    if (ls < 0)
    {
        perror("Socket creation failed");
        exit(1);
    }

    return ls;

void setup_server(int ls, struct sockaddr_in *servAddr)
{
    int opt = 1;

    setsockopt(ls, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    memset(servAddr, 0, sizeof(*servAddr));

    servAddr->sin_family = AF_INET;

    // All available network interfaces
    servAddr->sin_addr.s_addr = htonl(INADDR_ANY);

    servAddr->sin_port = htons(SERV_PORT);

    if (bind(ls, (struct sockaddr *)servAddr,
             sizeof(*servAddr)) < 0)
    {
        perror("Binding failed");
        close(ls);
        exit(1);
    }

    if (listen(ls, 5) < 0)
    {
        perror("Listening failed");
        close(ls);
        exit(1);
    }
}

int accept_client(int ls, struct sockaddr_in *clientAddr,
                  socklen_t *clientAddrLen)
{
    int s;

    *clientAddrLen = sizeof(*clientAddr);

    s = accept(ls, (struct sockaddr *)clientAddr,
               clientAddrLen);

    if (s < 0)
    {
        perror("Accept failed");
        close(ls);
        exit(1);
    }

    return s;
}

        perror("Receiving array failed");
        exit(1);

    return n;
