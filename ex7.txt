#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>
#include <netinet/in.h>

#define PORT 8080
#define BUFFER_SIZE 1024
#define MAX_CLIENTS 100


/*========================================
        CLIENT STRUCTURE
========================================*/

struct Client
{
    struct sockaddr_in address;
    int id;
    int active;
    int waiting;
    int finished;
};


/*========================================
        FUNCTION DECLARATIONS
========================================*/

int createSocket();

void bindSocket(int serverSocket,
                struct sockaddr_in *serverAddr);

int findClient(struct Client clients[],
               int clientCount,
               struct sockaddr_in *clientAddr);

int addClient(struct Client clients[],
              int *clientCount,
              struct sockaddr_in *clientAddr);

void sendMessage(int serverSocket,
                 char message[],
                 struct sockaddr_in *clientAddr);

int findNextClient(struct Client clients[],
                   int clientCount);

void chat(int serverSocket);


/*========================================
        CREATE SOCKET
========================================*/

int createSocket()
{
    int sock;

    sock = socket(AF_INET, SOCK_DGRAM, 0);

    if(sock < 0)
    {
        perror("Socket creation failed");
        exit(1);
    }

    printf("UDP Server socket created successfully\n");

    return sock;
}


/*========================================
        BIND SOCKET
========================================*/

void bindSocket(int serverSocket,
                struct sockaddr_in *serverAddr)
{
    int option = 1;

    setsockopt(serverSocket,
               SOL_SOCKET,
               SO_REUSEADDR,
               &option,
               sizeof(option));


    memset(serverAddr,
           0,
           sizeof(*serverAddr));


    serverAddr->sin_family = AF_INET;

    serverAddr->sin_addr.s_addr =
        INADDR_ANY;

    serverAddr->sin_port =
        htons(PORT);


    if(bind(serverSocket,
            (struct sockaddr *)serverAddr,
            sizeof(*serverAddr)) < 0)
    {
        perror("Bind failed");

        close(serverSocket);

        exit(1);
    }


    printf("Server is bound to port %d\n",
           PORT);
}


/*========================================
        FIND EXISTING CLIENT
========================================*/

int findClient(struct Client clients[],
               int clientCount,
               struct sockaddr_in *clientAddr)
{
    int i;

    for(i = 0; i < clientCount; i++)
    {
        if(clients[i].address.sin_addr.s_addr ==
           clientAddr->sin_addr.s_addr &&
           clients[i].address.sin_port ==
           clientAddr->sin_port)
        {
            return i;
        }
    }

    return -1;
}


/*========================================
        ADD NEW CLIENT
========================================*/

int addClient(struct Client clients[],
              int *clientCount,
              struct sockaddr_in *clientAddr)
{
    int index;

    if(*clientCount >= MAX_CLIENTS)
        return -1;


    index = *clientCount;


    clients[index].address =
        *clientAddr;

    clients[index].id =
        index + 1;

    clients[index].active = 1;

    clients[index].waiting = 1;

    clients[index].finished = 0;


    (*clientCount)++;


    return index;
}


/*========================================
        SEND MESSAGE
========================================*/

void sendMessage(int serverSocket,
                 char message[],
                 struct sockaddr_in *clientAddr)
{
    sendto(serverSocket,
           message,
           strlen(message),
           0,
           (struct sockaddr *)clientAddr,
           sizeof(*clientAddr));
}


/*========================================
        FIND NEXT WAITING CLIENT
========================================*/

int findNextClient(struct Client clients[],
                   int clientCount)
{
    int i;

    for(i = 0; i < clientCount; i++)
    {
        if(clients[i].active == 1 &&
           clients[i].waiting == 1 &&
           clients[i].finished == 0)
        {
            return i;
        }
    }

    return -1;
}


/*========================================
              CHAT
========================================*/

void chat(int serverSocket)
{
    struct Client clients[MAX_CLIENTS];

    int clientCount = 0;

    int currentClient = -1;

    int totalFinished = 0;


    char buffer[BUFFER_SIZE];

    char reply[BUFFER_SIZE];


    struct sockaddr_in clientAddr;

    socklen_t clientAddrLen;


    while(1)
    {
        /*
         * Receive message from any client
         */

        clientAddrLen =
            sizeof(clientAddr);


        memset(buffer,
               0,
               sizeof(buffer));


        int bytesReceived =
            recvfrom(serverSocket,
                     buffer,
                     BUFFER_SIZE - 1,
                     0,
                     (struct sockaddr *)&clientAddr,
                     &clientAddrLen);


        if(bytesReceived < 0)
        {
            perror("Receive failed");
            continue;
        }


        buffer[bytesReceived] = '\0';


        /*====================================
             CHECK WHETHER CLIENT EXISTS
        ====================================*/

        int index =
            findClient(clients,
                       clientCount,
                       &clientAddr);


        /*====================================
              NEW CLIENT
        ====================================*/

        if(index == -1)
        {
            index =
                addClient(clients,
                          &clientCount,
                          &clientAddr);


            if(index == -1)
            {
                strcpy(reply,
                       "Server full.");

                sendMessage(serverSocket,
                            reply,
                            &clientAddr);

                continue;
            }


            printf("\nClient %d connected.\n",
                   clients[index].id);


            /*
             * If no client is currently chatting,
             * this client becomes active.
             */

            if(currentClient == -1)
            {
                currentClient = index;

                clients[index].waiting = 0;

                strcpy(reply,
                       "START");

                sendMessage(serverSocket,
                            reply,
                            &clientAddr);


                printf("Client %d is now chatting with server.\n",
                       clients[index].id);
            }
            else
            {
                /*
                 * Another client is already chatting.
                 */

                strcpy(reply,
                       "WAIT");

                sendMessage(serverSocket,
                            reply,
                            &clientAddr);


                printf("Client %d is waiting.\n",
                       clients[index].id);
            }


            continue;
        }


        /*====================================
          CLIENT IS NOT CURRENTLY ACTIVE
        ====================================*/

        if(index != currentClient)
        {
            /*
             * Client may send another CONNECT
             * while waiting.
             */

            if(strcmp(buffer, "CONNECT") == 0)
            {
                strcpy(reply, "WAIT");

                sendMessage(serverSocket,
                            reply,
                            &clients[index].address);
            }

            continue;
        }


        /*====================================
              CURRENT CLIENT MESSAGE
        ====================================*/

        printf("\nClient %d: %s\n",
               clients[index].id,
               buffer);


        /*====================================
                CLIENT SAYS BYE
        ====================================*/

        if(strcmp(buffer, "bye") == 0)
        {
            strcpy(reply, "bye");

            sendMessage(serverSocket,
                        reply,
                        &clients[index].address);


            printf("Client %d disconnected.\n",
                   clients[index].id);


            clients[index].active = 0;

            clients[index].waiting = 0;

            clients[index].finished = 1;


            currentClient = -1;

            totalFinished++;


            /*
             * Check for next waiting client
             */

            currentClient =
                findNextClient(clients,
                               clientCount);


            if(currentClient != -1)
            {
                clients[currentClient].waiting = 0;


                strcpy(reply,
                       "START");

                sendMessage(
                    serverSocket,
                    reply,
                    &clients[currentClient].address
                );


                printf("Client %d is now chatting with server.\n",
                       clients[currentClient].id);
            }


            /*
             * All clients finished
             */

            if(totalFinished == clientCount)
            {
                printf("\nAll clients have finished.\n");

                break;
            }


            continue;
        }


        /*====================================
              SERVER REPLY
        ====================================*/

        printf("Reply to Client %d: ",
               clients[index].id);

        fflush(stdout);


        fgets(reply,
              BUFFER_SIZE,
              stdin);


        reply[strcspn(reply, "\n")] = '\0';


        sendMessage(
            serverSocket,
            reply,
            &clients[index].address
        );


        /*====================================
             SERVER ALSO SAYS BYE
        ====================================*/

        if(strcmp(reply, "bye") == 0)
        {
            printf("Chat ended with Client %d.\n",
                   clients[index].id);


            clients[index].active = 0;

            clients[index].waiting = 0;

            clients[index].finished = 1;


            currentClient = -1;

            totalFinished++;


            currentClient =
                findNextClient(clients,
                               clientCount);


            if(currentClient != -1)
            {
                clients[currentClient].waiting = 0;


                strcpy(reply,
                       "START");


                sendMessage(
                    serverSocket,
                    reply,
                    &clients[currentClient].address
                );


                printf("Client %d is now chatting with server.\n",
                       clients[currentClient].id);
            }


            if(totalFinished == clientCount)
            {
                printf("\nAll clients have finished.\n");

                break;
            }
        }
    }
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {

typedef struct {

unsigned int ip_to_uint(const char *ip) {
    unsigned int a, b, c, d;

    if (sscanf(ip, "%u.%u.%u.%u", &a, &b, &c, &d) != 4) {
        return 0;
    }

    return (a << 24) | (b << 16) | (c << 8) | d;
}

void uint_to_ip(unsigned int ip, char *buffer) {
    sprintf(buffer,
            "%u.%u.%u.%u",
            (ip >> 24) & 255,
            (ip >> 16) & 255,
            (ip >> 8) & 255,
            ip & 255);
}

int next_power_of_two(int value) {
    int power = 1;

    while (power < value) {
        power = power * 2;
    }

    return power;
}

int power_number(int value) {
    int power = 0;

    while (value > 1) {
        value = value / 2;
        power++;
    }

    return power;
}

void create_subnet_pool(int hosts, int prefix, int subnet_count) {
    int address_block;
    int base_subnet_size;
    int remainder;
    int i;

    unsigned int base_network;
    unsigned int current_network;

    address_block = next_power_of_two(hosts);

    /*
     * Demo base network.
     * Prefix /24 means:
     * 192.168.1.0 to 192.168.1.255
     */
    base_network = ip_to_uint("192.168.1.0");
    current_network = base_network;

    base_subnet_size = address_block / subnet_count;
    remainder = address_block % subnet_count;

    /*
     * If subnet count is greater than address block,
     * give one address to each subnet.
     */
    if (base_subnet_size == 0) {
        base_subnet_size = 1;
        remainder = 0;
    }

    for (i = 0; i < subnet_count; i++) {
        int subnet_size;

        subnet_size = base_subnet_size;

        /*
         * Remaining addresses are added
         * to the last subnet.
         */
        if (i == subnet_count - 1) {
            subnet_size = subnet_size + remainder;
        }

        subnet_pool[i].network = current_network;
        subnet_pool[i].broadcast =
            current_network + subnet_size - 1;

        /*
         * Network IP and broadcast IP are reserved.
         */
        if (subnet_size >= 2) {
            subnet_pool[i].first_ip =
                subnet_pool[i].network + 1;

            subnet_pool[i].last_ip =
                subnet_pool[i].broadcast - 1;

            subnet_pool[i].next_ip =
                subnet_pool[i].first_ip;
        } else {
            subnet_pool[i].first_ip =
                subnet_pool[i].network;

            subnet_pool[i].last_ip =
                subnet_pool[i].network;

            subnet_pool[i].next_ip =
                subnet_pool[i].network;
        }

        subnet_pool[i].assigned = 0;

        current_network =
            current_network + subnet_size;
    }

    old_hosts = hosts;
    old_prefix = prefix;
    old_subnet_count = subnet_count;
    pool_initialized = 1;
}

int create_response(int subnet_no,
                    DHCPResponse *response) {
    SubnetInfo *subnet;

    char network[32];
    char assigned_ip[32];
    char last_ip[32];
    char broadcast[32];

    memset(response, 0, sizeof(DHCPResponse));

    if (subnet_no < 1 ||
        subnet_no > old_subnet_count) {
        strcpy(response->status, "INVALID SUBNET");
        return -1;
    }

    subnet = &subnet_pool[subnet_no - 1];

    if (subnet->next_ip > subnet->last_ip) {
        strcpy(response->status, "NO IP AVAILABLE");
        return -1;
    }

    uint_to_ip(subnet->network, network);
    uint_to_ip(subnet->next_ip, assigned_ip);
    uint_to_ip(subnet->last_ip, last_ip);
    uint_to_ip(subnet->broadcast, broadcast);

    strcpy(response->network, network);
    strcpy(response->assigned_ip, assigned_ip);
    strcpy(response->last_ip, last_ip);
    strcpy(response->broadcast, broadcast);
    strcpy(response->status, "SUCCESS");

    /*
     * Mark subnet as assigned.
     */
    subnet->assigned = 1;

    /*
     * Next client gets the next IP.
     */
    subnet->next_ip++;

    return 0;
}

        printf("[WAIT] Server is ready for the next client.\n");

        close(client_socket);
    }

    close(server_socket);

void processDNS(
    int clientSocket,
    struct sockaddr_in *serverAddr
)
{
    char domain[BUFFER_SIZE];

    char ip[BUFFER_SIZE];

    /*----------------------------------------------*/
    /* Wait For Client Turn */
    /*----------------------------------------------*/

    waitForTurn(
        clientSocket,
        serverAddr
    );

    /*----------------------------------------------*/
    /* DNS Query */
    /*----------------------------------------------*/

    while (1)
    {
        printf(
            "\nEnter domain name: "
        );

        scanf(
            "%1023s",
            domain
        );

        /* Send domain */

        sendMessage(
            clientSocket,
            domain,
            serverAddr
        );

        /* Receive IP */

        receiveMessage(
            clientSocket,
            ip
        );

        printf(
            "DNS Server Response: %s\n",
            ip
        );

        /* Exit */

        if (
            strcmp(
                domain,
                "exit"
            ) == 0
        )
        {
            printf(
                "DNS client terminated.\n"
            );

            break;
        }

        printf(
            "IP Address for %s = %s\n",
            domain,
            ip
        );
