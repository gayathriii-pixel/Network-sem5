void clearInputBuffer(void)

int searchURL(char url[])

void decimalToBinary(int num, char binary[])
{
    int i;
    for(i = 7; i >= 0; i--)
        binary[7 - i] = ((num >> i) & 1) + '0';
    binary[8] = '\0';
}

void ipToBinary(char ip[], char binary[])
{
    int a, b, c, d;
    char temp[9];
    binary[0] = '\0';
    sscanf(ip, "%d.%d.%d.%d", &a, &b, &c, &d);
    decimalToBinary(a, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(b, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(c, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(d, temp); strcat(binary, temp);
}

void macToBinary(char mac[], char binary[])
{
    unsigned int a, b, c, d, e, f;
    char temp[9];
    binary[0] = '\0';
    sscanf(mac, "%x:%x:%x:%x:%x:%x", &a, &b, &c, &d, &e, &f);
    decimalToBinary(a, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(b, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(c, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(d, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(e, temp); strcat(binary, temp); strcat(binary, " ");
    decimalToBinary(f, temp); strcat(binary, temp);
}

void textToBinary(char text[], char binary[])
{
    int i;
    char temp[9];
    binary[0] = '\0';
    for(i = 0; text[i] != '\0'; i++)
    {
        if(text[i] == '\n') break;
        decimalToBinary((int)text[i], temp);
        strcat(binary, temp);
    }
}

void createPacket(char binary[], char srcIPBin[], char destIPBin[], char srcMACBin[], char destMACBin[])
{
    int length = strlen(binary);
    int i, j;
    char packet[25];
    int packet_count = 0;
    int total_frames = 0;

    printf("\n========================================");
    printf("\nPACKET FORMATION (24 BITS)");
    printf("\n========================================\n");

    for(i = 0; i < length; i += 24)
    {
        packet_count++;
        memset(packet, '0', 24);
        packet[24] = '\0';

        for(j = 0; j < 24 && (i + j) < length; j++)
            packet[j] = binary[i + j];

        printf("\nPacket %d\n", packet_count);
        printf("Source IP      : %s\n", srcIPBin);
        printf("Destination IP : %s\n", destIPBin);
        printf("Binary Text    : %s\n", packet);

        createFrames(packet, srcIPBin, destIPBin, srcMACBin, destMACBin);
        total_frames += 3;

        printf("\n----------------------------------------\n");
    }

    printf("Total packets generated: %d\n", packet_count);
    printf("Total frames generated : %d\n", total_frames);
}

void createFrames(char packet[], char srcIPBin[], char destIPBin[], char srcMACBin[], char destMACBin[])
{
    int i, j;
    char frame[9];
    int frame_count = 0;

    printf("\n-------DATA LAYER-----------\n");
    printf("Source MAC      : %s\n", srcMACBin);
    printf("Destination MAC : %s\n", destMACBin);

    for(i = 0; i < 24; i += 8)
    {
        for(j = 0; j < 8; j++)
            frame[j] = packet[i + j];
        frame[8] = '\0';
        frame_count++;

        printf("Frame %d : %s\n", frame_count, frame);
    }

    printf("Total frames in this packet: %d\n", frame_count);
}
